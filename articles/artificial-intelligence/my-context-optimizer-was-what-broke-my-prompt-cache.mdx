   ```python
   # context_optimizer.py
   """Context optimizer for LLM prompt normalization and budgeting."""
   from __future__ import annotations
   from typing import Dict, List, Optional
   
   
   class ContextOptimizer:
       """Reshapes user prompts to improve model relevance while preserving semantics."""
   
       def __init__(self, entity_map: Dict[str, str], max_tokens: int):
           if max_tokens <= 0:
               raise ValueError("max_tokens must be a positive integer")
           self.entity_map = entity_map
           self.max_tokens = max_tokens
   
       def normalize(self, text: str) -> str:
           """Collapse whitespace, strip, and lower‑case for consistent processing."""
           if not isinstance(text, str):
               raise TypeError("text must be a string")
           # collapse any run of whitespace to a single space
           collapsed = " ".join(text.split())
           # strip leading/trailing whitespace (already handled by split/join)
           # case‑fold for case‑insensitive matching
           return collapsed.lower()
   
       def disambiguate(self, tokens: List[str]) -> List[str]:
           """Replace ambiguous tokens with canonical IDs using the entity lookup."""
           resolved: List[str] = []
           for token in tokens:
               # use .get() to keep unknown tokens unchanged
               resolved.append(self.entity_map.get(token, token))
           return resolved
   
       def budget(self, tokens: List[str]) -> List[str]:
           """Enforce a hard token limit using a deterministic FIFO strategy."""
           if len(tokens) <= self.max_tokens:
               return tokens
           # keep the first N tokens; this preserves the opening context
           return tokens[: self.max_tokens]
   
       def optimize(self, prompt: str) -> str:
           """Run the full optimization pipeline on a user prompt."""
           if not prompt or not prompt.strip():
               return ""
           # naive tokenization by splitting on whitespace
           tokens = self.normalize(prompt).split()
           tokens = self.disambiguate(tokens)
           tokens = self.budget(tokens)
           return " ".join(tokens)
   ```
