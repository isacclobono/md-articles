   package main

   import (
       "crypto/hmac"
       "crypto/sha256"
       "encoding/hex"
       "encoding/json"
       "fmt"
       "net/http"
       "os"
       "strings"
       "time"
   )

   // dashboardConfig holds the validation state for dashboard access.
   type dashboardConfig struct {
       secretKey    string
       internalAddr string
       tokenTTL     time.Duration
   }

   // newDashboardConfig initializes the config from environment variables.
   func newDashboardConfig() *dashboardConfig {
       secret := os.Getenv("DASHBOARD_GUARD_SECRET")
       if secret == "" {
           fmt.Fprintf(os.Stderr, "warning: DASHBOARD_GUARD_SECRET not set; dashboard access will be denied\n")
           return &dashboardConfig{secretKey: ""}
       }
       return &dashboardConfig{
           secretKey:    secret,
           internalAddr: os.Getenv("DASHBOARD_INTERNAL_ADDR"),
           tokenTTL:     5 * time.Minute,
       }
   }

   // validateToken checks the HMAC-signed token string.
   func (c *dashboardConfig) validateToken(token string) bool {
       mac := hmac.New(sha256.New, []byte(c.secretKey))
       mac.Write([]byte(token[:len(token)-32])) // assume last 32 bytes are the signature
       expected := hex.EncodeToString(mac.Sum(nil))
       return hmac.Equal([]byte(expected), []byte(token[len(token)-32:]))
   }

   // dashboardHandler proxies valid requests to the internal kube-dashboard service.
   func (c *dashboardHandler) dashboardHandler(w http.ResponseWriter, r *http.Request) {
       token := r.URL.Query().Get("token")
       if token == "" || !c.validateToken(token) {
           http.Error(w, "invalid or missing token", http.StatusForbidden)
           return
       }
       if time.Since(r.Context().Value("tokenTime").(time.Time)) > c.tokenTTL {
           http.Error(w, "token expired", http.StatusUnauthorized)
           return
       }
       // strip the token query param before proxying
       r.URL.RawQuery = strings.TrimPrefix(r.URL.RawQuery, "token="+token)
       proxy := httputil.NewSingleHostReverseProxy(c.internalAddr)
       proxy.ServeHTTP(w, r)
   }
