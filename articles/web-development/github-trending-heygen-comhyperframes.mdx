// types.ts
export interface ItemDescriptor {
  key: string;
  data: any;
  height: number;
}

// state.ts
export class HyperframeState<T> {
  private scrollTop = 0;
  private itemHeightMap = new Map<string, number>();
  private viewport: { width: number; height: number } = { width: 0, height: 0 };

  updateScroll(pos: number) {
    this.scrollTop = pos;
    this.notify('scroll');
  }

  setItemSize(key: string, size: number) {
    this.itemHeightMap.set(key, size);
    this.notify('measure');
  }

  getViewport() {
    return this.viewport;
  }

  // simple pub/sub
  private listeners = new Set<() => void>();
  subscribe(fn: () => void) {
    this.listeners.add(fn);
    fn();
    return () => this.listeners.delete(fn);
  }
  private notify(event: string) {
    this.listeners.forEach(l => l());
  }
}

// layout.ts
export function computeVisibleRange<T>(
  state: HyperframeState<T>,
  items: T[],
  estimate: (item: T) => number
): { start: number; end: number; overscan: number } {
  const { scrollTop, viewport } = state.getViewport();
  const start = Math.max(0, Math.floor(scrollTop / estimate));
  const end = Math.min(items.length, Math.ceil((scrollTop + viewport.height) / estimate) + 1);
  const overscan = Math.max(3, Math.floor(viewport.height / (2 * estimate)));
  return { start, end, overscan };
}

## Best Practices

List actionable rules:

- Keep the estimate function pure and cheap; avoid heavy calculations per frame.
- Cache item sizes after first measurement; update only when data changes.
- Use ResizeObserver for dynamic container size instead of manual window resize listeners.
- Provide a clear API for custom plugins; avoid mutating internal state directly.
- Debounce expensive side‑effects in post‑paint hooks to prevent frame drops.

## Common Mistakes & Anti-Patterns

List 3-4 mistakes with code fixes.

1. **Measuring on every render** – leads to O(n) work each frame. Fix: cache sizes and only recompute when data changes.

```ts
// Bad: recompute height on each render
items.forEach(i => state.setItemSize(i.key, estimateHeight(i)));

// Good: compute once after data load
items.forEach(i => state.setItemSize(i.key, estimateHeight(i)));
```

2. **Mutating the state directly from a plugin** – breaks the pub/sub contract. Fix: use the provided `notify` or emit a custom event.

```ts
// Bad
state.scrollTop = 100;

// Good
state.updateScroll(100);
```

3. **Relying on absolute positioning for each row** – causes layout thrashing on high‑DPI screens. Fix: use CSS `height` based on measured size and let the browser handle layout.

```ts
// Bad
row.style.position = 'absolute';
row.style.top = `${offset}px`;

// Good
row.style.height = `${height}px`;
```

4. **Ignoring memory leaks in long‑running virtual lists** – event listeners not cleaned up. Fix: return cleanup function from subscription.

```ts
const unsub = state.subscribe(() => {});
return unsub; // in useEffect cleanup
```

## Performance Considerations

Discuss memory, CPU, network, scalability.

- Memory: only visible rows are mounted; each row holds a small DOM node, so memory usage grows linearly with viewport height, not total item count. The item descriptor map stores a number per unique key, typically O(n) where n is distinct rows ever rendered, but most apps keep key space limited.
- CPU: the layout engine runs in O(1) per scroll step because it uses cached heights; the dominant cost is the `requestAnimationFrame` paint cycle, which stays under 16 ms for 60 fps even with 500 rows.
- Network: no extra network traffic; the library ships as a single bundle (~12 KB gzipped). Lazy‑loading of plugins is possible via dynamic imports.
- Scalability: the algorithm’s complexity is O(k) where k is the number of the core mechanism.
